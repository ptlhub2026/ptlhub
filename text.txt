local ScriptCards = {}

local function CreateScript(name, description, callback)

    local Card = Instance.new("Frame")
    Card.Name = name
    Card.Size = UDim2.new(1, -5, 0, 58)
    Card.BackgroundColor3 = Color3.fromRGB(23, 26, 24)
    Card.BorderSizePixel = 0
    Card.Parent = ScriptList

    Instance.new("UICorner", Card).CornerRadius = UDim.new(0, 11)

    local Name = Instance.new("TextLabel")
    Name.Size = UDim2.new(1, -100, 0, 25)
    Name.Position = UDim2.fromOffset(14, 7)
    Name.BackgroundTransparency = 1
    Name.Text = name
    Name.TextXAlignment = Enum.TextXAlignment.Left
    Name.TextColor3 = Color3.fromRGB(240,240,240)
    Name.TextSize = 14
    Name.Font = Enum.Font.GothamBold
    Name.Parent = Card

    local Description = Instance.new("TextLabel")
    Description.Size = UDim2.new(1, -100, 0, 18)
    Description.Position = UDim2.fromOffset(14, 31)
    Description.BackgroundTransparency = 1
    Description.Text = description
    Description.TextXAlignment = Enum.TextXAlignment.Left
    Description.TextColor3 = Color3.fromRGB(115,125,120)
    Description.TextSize = 11
    Description.Font = Enum.Font.Gotham
    Description.Parent = Card

    local Run = Instance.new("TextButton")
    Run.Size = UDim2.fromOffset(65, 32)
    Run.Position = UDim2.new(1, -78, 0.5, -16)
    Run.BackgroundColor3 = Color3.fromRGB(50, 220, 110)
    Run.Text = "RUN"
    Run.TextColor3 = Color3.fromRGB(10,30,18)
    Run.TextSize = 12
    Run.Font = Enum.Font.GothamBold
    Run.Parent = Card

    Instance.new("UICorner", Run).CornerRadius = UDim.new(0, 9)

    Run.MouseEnter:Connect(function()
        TweenService:Create(
            Run,
            TweenInfo.new(0.15),
            {BackgroundColor3 = Color3.fromRGB(80,255,140)}
        ):Play()
    end)

    Run.MouseLeave:Connect(function()
        TweenService:Create(
            Run,
            TweenInfo.new(0.15),
            {BackgroundColor3 = Color3.fromRGB(50,220,110)}
        ):Play()
    end)

    Run.MouseButton1Click:Connect(function()
        if callback then
            callback()
        end
    end)

    table.insert(ScriptCards, {
        frame = Card,
        name = string.lower(name)
    })
end

--==================================================
-- SEARCH FUNCTION
--==================================================

Search:GetPropertyChangedSignal("Text"):Connect(function()

    local query = string.lower(Search.Text)

    for _,data in ipairs(ScriptCards) do
        data.frame.Visible =
            query == "" or string.find(data.name, query, 1, true) ~= nil
    end

end)
